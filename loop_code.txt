n = int(input("how old r u? "))
sum = ((n * (n + 1)) / 2)
while (n <= 0):

    n = int(input("how old r u? "))

def getsum(n): 
    ni = 0
    for i in range(0, n + 1):
        ni += i
    return ni

if sum == getsum(n): 
   print ("verified")
